<script>
  const COLS = 25;
  const ROWS = 32;
  const TOTAL_CELLS = COLS * ROWS;

  const gridElement = document.getElementById('grid');
  const textInput = document.getElementById('textInput');
  const charCount = document.getElementById('charCount');

  function createGrid() {
    gridElement.innerHTML = '';
    for (let i = 0; i < TOTAL_CELLS; i++) {
      const cell = document.createElement('div');
      cell.className = 'cell';
      gridElement.appendChild(cell);
    }
  }

  function updateGrid() {
    const text = textInput.value;
    const cells = gridElement.children;

    for (let i = 0; i < TOTAL_CELLS; i++) {
      cells[i].textContent = '';
    }

    let cellIndex = 0;
    for (let i = 0; i < text.length; i++) {
      if (cellIndex >= TOTAL_CELLS) break;
      const char = text[i];
      if (char === '\n') {
        const remainder = cellIndex % COLS;
        if (remainder !== 0) cellIndex += (COLS - remainder);
      } else {
        cells[cellIndex].textContent = char;
        cellIndex++;
      }
    }
    charCount.textContent = Math.min(text.replace(/\n/g, '').length, TOTAL_CELLS);
  }

  // タブ切り替え機能（スマホ用）
  function switchTab(tab) {
    const editSec = document.getElementById('editorSec');
    const prevSec = document.getElementById('previewSec');
    const tabEdit = document.getElementById('tabEdit');
    const tabPreview = document.getElementById('tabPreview');

    if (tab === 'edit') {
      editSec.classList.remove('hidden');
      prevSec.classList.add('hidden');
      tabEdit.classList.add('active');
      tabPreview.classList.remove('active');
    } else {
      editSec.classList.add('hidden');
      prevSec.classList.remove('hidden');
      tabEdit.classList.remove('active');
      tabPreview.classList.add('active');
    }
  }

  createGrid();
  textInput.addEventListener('input', updateGrid);
</script>
